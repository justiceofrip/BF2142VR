function(add_voice_opus)
 set(BUILD_TESTING OFF)
 set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
 set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
 set(OPUS_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
 set(OPUS_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
 add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/opus-1.6.1" "${CMAKE_CURRENT_BINARY_DIR}/opus" EXCLUDE_FROM_ALL)
endfunction()
add_voice_opus()
add_library(BF2142Voice STATIC VoiceProtocol.cpp VoiceCodec.cpp WaveAudio.cpp VoiceClient.cpp ../TrackingMath.cpp ../multiplayer/PoseProtocol.cpp)
target_include_directories(BF2142Voice PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
target_compile_definitions(BF2142Voice PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(BF2142Voice PRIVATE /W4 /permissive-)
set_property(TARGET BF2142Voice PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(BF2142Voice PUBLIC opus BFVRStereoMath winmm ws2_32 user32)
